🚀 COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. DEMO (Verify Setup)
   $ npm run seed:demo
   → 10 users, 100 posts, ~2 seconds

2. QUICK (Development)
   $ npm run seed:quick
   → 100 users, 5K posts, ~20 seconds

3. LARGE (Production Test)
   $ npm run seed:large
   → 5000 users, 10M posts, ~45 minutes ☕

4. CLEAN (Remove All Data)
   $ npm run seed:clean
   → Deletes everything (requires confirmation)


📋 TYPICAL WORKFLOWS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🔰 First Time Setup:
   $ npm run seed:demo          # Verify (1-2s)
   $ npm run dev                # Start server

💻 Daily Development:
   $ npm run seed:clean         # Clean data
   $ npm run seed:quick         # Load fresh data (15-30s)
   $ npm run dev                # Start server

🧪 Load Testing:
   $ npm run seed:clean         # Clean data
   $ npm run seed:large         # Load 10M posts (30-60m)
   # Run performance tests

⚡ PERFORMANCE TIPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

For Large Seeder:
   • Close other applications
   • Use SSD storage
   • Increase Node.js memory:
     NODE_OPTIONS="--max-old-space-size=4096" npm run seed:large


🐛 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Connection Error:
   → Check if MongoDB & Neo4j are running
   → npm run test-mongodb
   → npm run test-neo4j

Out of Memory:
   → Use seed:quick instead of seed:large
   → Increase Node.js memory limit

Duplicate Key Error:
   → Run: npm run seed:clean
   → Then run seeder again
